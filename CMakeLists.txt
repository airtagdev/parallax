cmake_minimum_required(VERSION 3.20)
project(Parallax VERSION 0.2.0 LANGUAGES C)
if(NOT WIN32)
  message(FATAL_ERROR "This first release targets Windows x64.")
endif()
add_library(parallax_core STATIC src/rewrite/json.c src/rewrite/compat.c)
target_include_directories(parallax_core PUBLIC src/rewrite)
set_property(TARGET parallax_core PROPERTY C_STANDARD 17)
target_compile_definitions(parallax_core PRIVATE _CRT_SECURE_NO_WARNINGS)
add_executable(Parallax WIN32 src/rewrite/main.c src/rewrite/resources.c src/rewrite/font.c src/rewrite/title.c)
set_property(TARGET Parallax PROPERTY C_STANDARD 17)
target_compile_definitions(Parallax PRIVATE _CRT_SECURE_NO_WARNINGS GL_BGR_EXT=0x80E0)
target_link_libraries(Parallax PRIVATE parallax_core opengl32 gdi32 user32 ole32 windowscodecs uuid)
if(MSVC)
  target_compile_options(Parallax PRIVATE /W4 /O2 /fp:strict)
  set_property(TARGET Parallax PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(parallax_core PRIVATE /W4 /O2 /fp:strict)
  set_property(TARGET parallax_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set_target_properties(Parallax PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/release")
enable_testing()
add_executable(ParallaxCoreTests tests/core_tests.c)
target_link_libraries(ParallaxCoreTests PRIVATE parallax_core)
set_property(TARGET ParallaxCoreTests PROPERTY C_STANDARD 17)
if(MSVC)
  set_property(TARGET ParallaxCoreTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_definitions(ParallaxCoreTests PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_test(NAME rewrite_core COMMAND ParallaxCoreTests)
add_executable(ParallaxImageCompare tests/image_compare.c)
target_link_libraries(ParallaxImageCompare PRIVATE ole32 windowscodecs uuid)
set_property(TARGET ParallaxImageCompare PROPERTY C_STANDARD 17)
if(MSVC)
  target_compile_definitions(ParallaxImageCompare PRIVATE _CRT_SECURE_NO_WARNINGS)
  set_property(TARGET ParallaxImageCompare PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/release/resources/meta/import.json")
  add_test(NAME imported_26_3_data COMMAND Parallax --verify-resources)
endif()
option(PARALLAX_BUILD_RETIRED_SANDBOX "Build the retired, non-vanilla 0.1 experiment" OFF)
if(PARALLAX_BUILD_RETIRED_SANDBOX)
  add_executable(ParallaxSandbox WIN32 experimental/sandbox/src/main.c experimental/sandbox/src/world.c experimental/sandbox/src/render.c)
  target_compile_definitions(ParallaxSandbox PRIVATE _CRT_SECURE_NO_WARNINGS GL_BGR_EXT=0x80E0)
  target_link_libraries(ParallaxSandbox PRIVATE opengl32 gdi32 user32 shell32 ole32 windowscodecs uuid winmm)
  set_property(TARGET ParallaxSandbox PROPERTY C_STANDARD 17)
endif()
