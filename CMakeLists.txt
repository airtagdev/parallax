cmake_minimum_required(VERSION 3.20)
project(Parallax VERSION 0.1.0 LANGUAGES C)
if(NOT WIN32)
  message(FATAL_ERROR "This first release targets Windows x64.")
endif()
add_executable(Parallax WIN32 src/main.c src/world.c src/render.c)
set_property(TARGET Parallax PROPERTY C_STANDARD 17)
target_compile_definitions(Parallax PRIVATE _CRT_SECURE_NO_WARNINGS GL_BGR_EXT=0x80E0)
target_link_libraries(Parallax PRIVATE opengl32 gdi32 user32 shell32 ole32 windowscodecs uuid winmm)
if(MSVC)
  target_compile_options(Parallax PRIVATE /W4 /O2 /fp:precise)
  set_property(TARGET Parallax PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set_target_properties(Parallax PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/release")
enable_testing()
add_test(NAME world_integrity COMMAND Parallax --self-test)
